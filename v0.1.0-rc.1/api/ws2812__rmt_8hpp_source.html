33// Optional hooks: a `Declares*` concept only tests the name, the `Has*` concept the exact
34// signature, `noexcept` and the result type; the driver rejects a hook that is declared but does
35// not match (handbook KB-CPP20-007).
36
38template <typename Port>
39concept DeclaresGpioSet =
40 requires(Port& port) { port.GpioSet(CtrlPin::Power, GpioSignal::ACTIVE); };
41
43template <typename Port>
44concept HasGpioSet =
45 requires { ExactHook<Port, bool(CtrlPin, GpioSignal) noexcept>::Take(&Port::GpioSet); };
46
48template <typename Port>
49concept DeclaresLog = requires(Port& port) { port.Log(LogLevel::Info, ""); };
50
52template <typename Port>
53concept HasLog =
54 requires { ExactHook<Port, void(LogLevel, const char*) noexcept>::Take(&Port::Log); };
55
57template <size_t N>
58concept NonZeroSize = (N > 0U);
59
61template <typename Port>
62void Log(Port& port, LogLevel level, const char* message) noexcept {
63 if constexpr (HasLog<Port>) {
64 port.Log(level, message);
65 } else {
66 static_cast<void>(port);
67 static_cast<void>(level);
68 static_cast<void>(message);
69 }
70}
