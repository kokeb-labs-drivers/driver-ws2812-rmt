108[[nodiscard]] constexpr bool IsValidOrder(const ColourOrder& order) noexcept {
109 if ((order.count != 3U) && (order.count != 4U)) {
110 return false;
111 }
112 uint32_t seen = 0U;
113 for (uint32_t i = 0U; i < static_cast<uint32_t>(order.count); ++i) {
114 const Channel c = order.ch[i];
115 if (!detail::IsEnumerator(c)) {
116 return false;
117 }
118 const uint32_t bit = uint32_t{1} << (static_cast<uint32_t>(detail::ToUnderlying(c)) & 31U);
119 if (((seen & bit) != 0U) || ((order.count == 3U) && (c == Channel::W))) {
120 return false;
121 }
122 seen |= bit;
123 }
124 return true;
125}

185[[nodiscard]] constexpr size_t DecodeWire(std::span<const uint8_t> wire, const ColourOrder& order,
186 std::span<Colour> out) noexcept {
187 if (!IsValidOrder(order)) {
188 return 0U;
189 }
190 const size_t width = (order.count == 4U) ? 4U : 3U; // IsValidOrder() allowed only 3 or 4
191 const size_t pixels = std::min(wire.size() / width, out.size());
192 for (size_t p = 0U; p < pixels; ++p) {
193 Colour colour{};
194 for (size_t i = 0U; i < width; ++i) {
195 const uint8_t v = wire[(p * width) + i];
196 switch (order.ch[i]) {
197 case Channel::R:
198 colour.r = v;
199 break;
200 case Channel::G:
201 colour.g = v;
202 break;
203 case Channel::B:
204 colour.b = v;
205 break;
206 case Channel::W:
207 colour.w = v;
208 break;
209 default:
210 break;
211 }
212 }
213 out[p] = colour;
214 }
215 return pixels;

339[[nodiscard]] constexpr bool ResolveBit(const Range& high, const Range& low, const Timing& t,
340 const TickBase& base, uint32_t& th, uint32_t& tl) noexcept {
341 const uint64_t hz = base.hz;
342 const uint64_t cap_ns = base.cap_ns;
343 const TickWindow hw = WindowOf(high, hz, cap_ns);
344 const TickWindow lw = WindowOf(low, hz, cap_ns);
345 if ((hw.lo > hw.hi) || (lw.lo > lw.hi) || (hw.hi > 0xFFFF'FFFFULL) || (lw.hi > 0xFFFF'FFFFULL)) {
346 return false;
347 }
348 const uint64_t h = Aim(high, hw, hz);
349 uint64_t l = Aim(low, lw, hz);
350 const uint64_t need = CeilTicks(t.period_min_ns, hz);
351 if ((h + l) < need) {
352 l = need - h;
353 }
354 const bool too_long =
355 (t.period_max_ns != 0U) && (((h + l) * kNsPerSecond) > (t.period_max_ns * hz));
356 if ((l > lw.hi) || too_long) {
357 return false;
358 }
359 th = static_cast<uint32_t>(h);
360 tl = static_cast<uint32_t>(l);
361 return true;
362}

367[[nodiscard]] constexpr bool InWindow(const Range& r, uint32_t ticks, uint64_t hz,
368 uint64_t cap_ns) noexcept {
369 const uint64_t scaled = static_cast<uint64_t>(ticks) * kNsPerSecond;
370 const bool above_min = scaled >= (static_cast<uint64_t>(r.min_ns) * hz);
371 bool below_max = true;
372 if (r.max_ns != 0U) {
373 below_max = scaled <= (static_cast<uint64_t>(r.max_ns) * hz);
374 } else if (cap_ns != 0U) {
375 below_max = scaled < (cap_ns * hz);
376 } else {
377 // No upper bound and no cap: nothing to check.
378 }
379 return above_min && below_max;

401[[nodiscard]] constexpr Result<Waveform> ResolveWaveform(const Timing& timing,
402 uint32_t resolution_hz,
403 uint32_t reset_us) noexcept {
404 using Failure = Result<Waveform>;
405 if ((resolution_hz == 0U) || (resolution_hz > kMaxResolutionHz) || (reset_us > kMaxResetUs) ||
406 !detail::TimingValid(timing)) {
407 return Failure::Failure(Error::TimingUnreachable);
408 }
409 const uint64_t hz = resolution_hz;
410 const auto cap_ns =
411 static_cast<uint64_t>((timing.max_low_ns != 0U) ? timing.max_low_ns : timing.reset_min_ns);
412 Waveform w{};
413 w.resolution_hz = resolution_hz;
414 const detail::TickBase base{.hz = hz, .cap_ns = cap_ns};
415 if (!detail::ResolveBit(timing.t0h, timing.t0l, timing, base, w.t0h_ticks, w.t0l_ticks) ||
416 !detail::ResolveBit(timing.t1h, timing.t1l, timing, base, w.t1h_ticks, w.t1l_ticks)) {
417 return Failure::Failure(Error::TimingUnreachable);
418 }
419 const uint64_t default_us =
420 ((static_cast<uint64_t>(timing.reset_min_ns) + 999U) / 1000U) + kResetMarginUs;
421 const uint64_t wanted_ns =
422 ((reset_us == 0U) ? default_us : static_cast<uint64_t>(reset_us)) * 1000U;
423 const uint64_t floor_ticks = detail::FloorTicks(timing.reset_min_ns, hz) + 1U;
424 w.reset_ticks = static_cast<uint32_t>(std::max(NsToTicks(wanted_ns, hz), floor_ticks));
425 // The longest low inside a frame: the smallest latch time any document of the variant prints,
426 // less 10 percent for the pixel's own threshold spread, and strictly below that.
427 const uint64_t gap_ns = cap_ns - (cap_ns / 10U);
428 w.max_gap_ticks = static_cast<uint32_t>(detail::CeilTicks(gap_ns, hz) - 1U);
429 // The driver's own lows must obey the limit it hands the port.
430 if ((w.t0l_ticks > w.max_gap_ticks) || (w.t1l_ticks > w.max_gap_ticks)) {
431 return Failure::Failure(Error::TimingUnreachable);
432 }
433 return Failure::Success(w);
434}

459[[nodiscard]] constexpr bool WaveformInWindow(Variant variant, const Waveform& wave) noexcept {
460 if ((variant == Variant::Custom) || !detail::IsEnumerator(variant) ||
461 (wave.resolution_hz == 0U)) {
462 return false;
463 }
464 const Timing t = ProfileTiming(variant);
465 const uint64_t hz = wave.resolution_hz;
466 const auto cap_ns = static_cast<uint64_t>((t.max_low_ns != 0U) ? t.max_low_ns : t.reset_min_ns);
467 const bool windows = detail::InWindow(t.t0h, wave.t0h_ticks, hz, 0U) &&
468 detail::InWindow(t.t0l, wave.t0l_ticks, hz, cap_ns) &&
469 detail::InWindow(t.t1h, wave.t1h_ticks, hz, 0U) &&
470 detail::InWindow(t.t1l, wave.t1l_ticks, hz, cap_ns);
471 const uint64_t p0 = (static_cast<uint64_t>(wave.t0h_ticks) + wave.t0l_ticks) * kNsPerSecond;
472 const uint64_t p1 = (static_cast<uint64_t>(wave.t1h_ticks) + wave.t1l_ticks) * kNsPerSecond;
473 const uint64_t pmin = static_cast<uint64_t>(t.period_min_ns) * hz;
474 const uint64_t pmax = static_cast<uint64_t>(t.period_max_ns) * hz;
475 const bool periods_min = (p0 >= pmin) && (p1 >= pmin);
476 const bool periods_max = (t.period_max_ns == 0U) || ((p0 <= pmax) && (p1 <= pmax));
477 const bool reset_ok = (static_cast<uint64_t>(wave.reset_ticks) * kNsPerSecond) >
478 (static_cast<uint64_t>(t.reset_min_ns) * hz);
479 return windows && periods_min && periods_max && reset_ok;
480}

575[[nodiscard]] constexpr Result<uint32_t> ResolveRmtRate(uint32_t source_hz, uint32_t requested_hz,
576 uint32_t tolerance_ppm = 1000U) noexcept {
577 if ((requested_hz == 0U) || (source_hz == 0U)) {
578 return Result<uint32_t>::Failure(Error::TimingUnreachable);
579 }
580 const uint32_t real = detail::RmtRealRate(source_hz, requested_hz);
581 const uint64_t deviation = (real > requested_hz) ? static_cast<uint64_t>(real - requested_hz)
582 : static_cast<uint64_t>(requested_hz - real);
583 if ((real == 0U) ||
584 ((deviation * 1'000'000ULL) > (static_cast<uint64_t>(tolerance_ppm) * requested_hz))) {
585 return Result<uint32_t>::Failure(Error::TimingUnreachable);
586 }
587 return Result<uint32_t>::Success(real);
