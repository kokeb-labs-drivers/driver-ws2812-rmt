243 [[nodiscard]] static constexpr Colour FromPacked(uint32_t rgb) noexcept {
244 return Colour{.r = static_cast<uint8_t>((rgb >> 16U) & 0xFFU),
245 .g = static_cast<uint8_t>((rgb >> 8U) & 0xFFU),

254 [[nodiscard]] static constexpr Colour FromPackedW(uint32_t wrgb) noexcept {
255 return Colour{.r = static_cast<uint8_t>((wrgb >> 16U) & 0xFFU),

257 .b = static_cast<uint8_t>(wrgb & 0xFFU),

262 return (static_cast<uint32_t>(r) << 16U) | (static_cast<uint32_t>(g) << 8U) |
