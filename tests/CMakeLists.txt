# =============================================================================
# WS2812_RMT driver: host tests
# =============================================================================
# The driver against a model of the RMT transmit path and a chain of pixels, with the firmware's flags
# (-fno-exceptions -fno-rtti) and warnings as errors. No hardware, no ESP-IDF:
#
#   cmake -S . -B build -DWS2812_RMT_BUILD_TESTS=ON
#   cmake --build build
#   ctest --test-dir build --output-on-failure
# =============================================================================

# ISO C++20, never the GNU dialect (handbook KB-CPP20-001): set here too, so that this folder is strict
# whoever includes it.
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# -pedantic-errors and the C++20 deprecations as errors (KB-CPP20-001, -003). -Wswitch-enum: a switch over
# an enumeration ends with `default: break;` (MISRA C++:2023 Rule 9.4.2), and a new enumerator still warns.
set(WS2812_RMT_TEST_FLAGS -fno-exceptions -fno-rtti -Wall -Wextra -Wpedantic -pedantic-errors -Wshadow
    -Wconversion -Wsign-conversion -Wdeprecated -Wdeprecated-enum-enum-conversion
    -Wdeprecated-enum-float-conversion -Wimplicit-fallthrough -Wswitch-enum -Werror)
# The standard library's own bounds checks, so an out-of-range span or array access traps in the tests
# and fuzzers (KB-CPP20-043).
set(WS2812_RMT_TEST_DEFINITIONS _GLIBCXX_ASSERTIONS=1 _LIBCPP_HARDENING_MODE=_LIBCPP_HARDENING_MODE_FAST)

function(ws2812_rmt_test_target name)
  target_link_libraries(${name} PRIVATE ws2812_rmt::ws2812_rmt)
  target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_features(${name} PRIVATE cxx_std_20)
  target_compile_options(${name} PRIVATE ${WS2812_RMT_TEST_FLAGS})
  target_compile_definitions(${name} PRIVATE ${WS2812_RMT_TEST_DEFINITIONS})
endfunction()

# --- Strict-dialect header checks ---------------------------------------------
# One translation unit per public header (and through the driver header its src/ .ipp): each header
# compiles on its own, first in its unit, as ISO C++20 with -pedantic-errors, and the unit asserts
# that the dialect really is C++20 (handbook KB-CPP20-001, checks strict-dialect and
# header-self-contained). They build with the host tests on GCC and Clang and, as part of
# ws2812_rmt_optimised_build, in the Cortex-M cross-build.
file(GLOB_RECURSE WS2812_RMT_PUBLIC_HEADERS CONFIGURE_DEPENDS ${PROJECT_SOURCE_DIR}/inc/*.hpp
  ${PROJECT_SOURCE_DIR}/inc/*.h)
set(WS2812_RMT_HEADER_CHECKS)
foreach(header ${WS2812_RMT_PUBLIC_HEADERS} ${WS2812_RMT_VERSION_HEADER})
  if(header STREQUAL WS2812_RMT_VERSION_HEADER)
    file(RELATIVE_PATH rel ${WS2812_RMT_VERSION_HEADER_DIR} ${header})
  else()
    file(RELATIVE_PATH rel ${PROJECT_SOURCE_DIR}/inc ${header})
  endif()
  string(MAKE_C_IDENTIFIER "${rel}" id)
  set(check ${CMAKE_CURRENT_BINARY_DIR}/header_checks/${id}.cpp)
  string(CONCAT content "// Generated by tests/CMakeLists.txt: ${rel} compiles on its own as ISO C++20.\n"
    "#include \"${rel}\"\n"
    "static_assert(__cplusplus == 202002L, \"the strict-dialect check must build as -std=c++20\");\n")
  if(EXISTS ${check})
    file(READ ${check} old)
  else()
    set(old "")
  endif()
  if(NOT old STREQUAL content)
    file(WRITE ${check} "${content}")
  endif()
  list(APPEND WS2812_RMT_HEADER_CHECKS ${check})
endforeach()
add_library(ws2812_rmt_header_checks OBJECT ${WS2812_RMT_HEADER_CHECKS})
ws2812_rmt_test_target(ws2812_rmt_header_checks)

# --- ESP32 port and examples on the host, over stand-ins of the ESP-IDF headers ---
# Each example defines app_main(), so each gets its own executable; the tests call the example's own
# Start(), the start-up app_main() runs (RD-21). The ESP32-H2 build checks the target's default rate.
function(ws2812_rmt_esp32_test name example)
  add_executable(${name} test_main.cpp ${ARGN}
    ${PROJECT_SOURCE_DIR}/examples/esp32/main/${example}.cpp)
  ws2812_rmt_test_target(${name})
  target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/esp32_stub
    ${PROJECT_SOURCE_DIR}/examples/esp32/main)
  add_test(NAME ${name} COMMAND ${name})
endfunction()
ws2812_rmt_esp32_test(ws2812_rmt_esp32_basic ws2812_rmt_basic_example esp32_stub/esp32_basic_test.cpp)
ws2812_rmt_esp32_test(ws2812_rmt_esp32_custom ws2812_rmt_custom_example esp32_stub/esp32_custom_test.cpp)
ws2812_rmt_esp32_test(ws2812_rmt_esp32_h2 ws2812_rmt_basic_example esp32_stub/esp32_h2_test.cpp)
target_compile_definitions(ws2812_rmt_esp32_h2 PRIVATE CONFIG_IDF_TARGET_ESP32H2=1)

# --- Version: the getters, and every copy of the version string ---------------
function(ws2812_rmt_version_test name include_dir major minor patch)
  add_executable(${name} test_version.cpp)
  if(include_dir)
    target_include_directories(${name} BEFORE PRIVATE ${include_dir})
  endif()
  ws2812_rmt_test_target(${name})
  target_compile_definitions(${name} PRIVATE EXPECTED_MAJOR=${major} EXPECTED_MINOR=${minor}
    EXPECTED_PATCH=${patch} EXPECTED_STRING="${major}.${minor}.${patch}")
  add_test(NAME ${name} COMMAND ${name})
endfunction()

ws2812_rmt_version_test(ws2812_rmt_version "" ${PROJECT_VERSION_MAJOR} ${PROJECT_VERSION_MINOR}
  ${PROJECT_VERSION_PATCH})

# The same template with another version, found ahead of the real header.
function(ws2812_rmt_version_probe_header)
  set(WS2812_RMT_VERSION_MAJOR 9)
  set(WS2812_RMT_VERSION_MINOR 8)
  set(WS2812_RMT_VERSION_PATCH 7)
  set(WS2812_RMT_VERSION 9.8.7)
  configure_file(${WS2812_RMT_VERSION_TEMPLATE} ${CMAKE_CURRENT_BINARY_DIR}/version_probe/ws2812_rmt_version.h @ONLY)
endfunction()
ws2812_rmt_version_probe_header()
ws2812_rmt_version_test(ws2812_rmt_version_probe ${CMAKE_CURRENT_BINARY_DIR}/version_probe 9 8 7)

# Files that carry their own copy of the version must match cmake/ws2812_rmt_build_settings.cmake.

  string(REPLACE "|" ";" parts "${copy}")
  list(GET parts 0 file)
  list(GET parts 1 key)
  string(MAKE_C_IDENTIFIER "${file}" id)
  add_test(NAME ws2812_rmt_version_copy_${id}
    COMMAND ${CMAKE_COMMAND} -DFILE=${PROJECT_SOURCE_DIR}/${file} -DKEY=${key} -DVERSION=${PROJECT_VERSION}
            -P ${CMAKE_CURRENT_SOURCE_DIR}/check_version_copy.cmake)
endforeach()

# --- Compile-time contract: these must fail to build, for the reason in their EXPECT: line ---
file(GLOB WS2812_RMT_COMPILE_FAIL CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/compile_fail/*.cpp)
foreach(src ${WS2812_RMT_COMPILE_FAIL})
  get_filename_component(stem ${src} NAME_WE)
  file(STRINGS ${src} expect REGEX "EXPECT: ")
  string(REGEX REPLACE ".*EXPECT: " "" expect "${expect}")
  add_executable(ws2812_rmt_compile_fail_${stem} EXCLUDE_FROM_ALL ${src})
  ws2812_rmt_test_target(ws2812_rmt_compile_fail_${stem})
  add_test(NAME ws2812_rmt_compile_fail_${stem}
    COMMAND ${CMAKE_COMMAND} --build ${CMAKE_BINARY_DIR} --target ws2812_rmt_compile_fail_${stem})
  # Pass only when the compiler rejects the code with the expected diagnostic, not another error.
  set_tests_properties(ws2812_rmt_compile_fail_${stem} PROPERTIES PASS_REGULAR_EXPRESSION "${expect}")
endforeach()
